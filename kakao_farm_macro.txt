// 플로팅 윈도우 드래그 이동 설정
let touchX, touchY, winX, winY, isMove = false;
window.statusText.setOnTouchListener(function(v, e) {
    if (e.getAction() == e.ACTION_DOWN) {
        touchX = e.getRawX(); touchY = e.getRawY();
        winX = window.getX(); winY = window.getY();
        isMove = true; return true;
    } else if (e.getAction() == e.ACTION_MOVE && isMove) {
        window.setPosition(winX + (e.getRawX() - touchX), winY + (e.getRawY() - touchY));
        return true;
    } else if (e.getAction() == e.ACTION_UP) {
        isMove = false; return true;
    }
    return false;
});

// [1] 상자 깨기 (상자 전 영역 고속 연타)
function breakBoxes() {
    // 밭 전체를 격자 형태로 빠르게 탭하여 나타난 상자를 모두 부숨
    let cols = 4;
    let rows = 4;
    let stepX = (farmArea.right - farmArea.left) / cols;
    let stepY = (farmArea.bottom - farmArea.top) / rows;

    for (let r = 0; r < rows; r++) {
        for (let c = 0; c < cols; c++) {
            if (!isRunning) return;
            let cx = farmArea.left + (c + 0.5) * stepX + random(-10, 10);
            let cy = farmArea.top + (r + 0.5) * stepY + random(-10, 10);
            // 연속 2회 탭 (상자 개봉)
            press(cx, cy, 30);
            sleep(40);
            press(cx, cy, 30);
            sleep(40);
        }
    }
}

// [2] OCR 화면 캡처 및 번호(레벨) 인식 후 동일 숫자 합성
function mergeSameNumbers() {
    let img = captureScreen();
    if (!img) return;

    // 농장 밭 영역만 이미지 크롭
    let cropImg = images.clip(img, farmArea.left, farmArea.top, farmArea.right - farmArea.left, farmArea.bottom - farmArea.top);
    
    // OCR 텍스트 감지 (AutoX 내장 gmlkit / paddle OCR 사용)
    let ocrResults = [];
    try {
        if (typeof gmlkit !== 'undefined' && gmlkit.ocr) {
            ocrResults = gmlkit.ocr(cropImg, "zh").toArray();
        } else if (typeof paddle !== 'undefined' && paddle.ocr) {
            ocrResults = paddle.ocr(cropImg);
        }
    } catch (e) {
        // OCR 에러 시 대비
    }

    cropImg.recycle();

    // 감지된 숫자들 그룹화 (예: "1", "2", "3" ...)
    let numberMap = {};
    if (ocrResults && ocrResults.length > 0) {
        for (let item of ocrResults) {
            let txt = ("" + (item.text || "")).replace(/[^0-9]/g, "");
            if (txt.length >= 1 && txt.length <= 3) { // 1~3자리 숫자
                let b = item.bounds;
                let actualX = farmArea.left + (b.left + b.right) / 2;
                let actualY = farmArea.top + (b.top + b.bottom) / 2;

                if (!numberMap[txt]) numberMap[txt] = [];
                numberMap[txt].push({ x: actualX, y: actualY });
            }
        }
    }

    // 동일한 번호가 2개 이상 있는 작물 찾아 드래그 합성
    let mergedCount = 0;
    for (let num in numberMap) {
        let list = numberMap[num];
        while (list.length >= 2) {
            if (!isRunning) return;
            let itemA = list.pop();
            let itemB = list.pop();

            // A를 B로 드래그(Swipe)해서 합성!
            swipe(itemA.x, itemA.y, itemB.x, itemB.y, 280);
            sleep(350);
            mergedCount++;
        }
    }

    return mergedCount;
}

// [3] 스마트 통합 루프 (상자 깨기 ➔ 작물 심기 ➔ 같은 번호 감지 합성)
function startSmartLoop() {
    isRunning = true;
    ui.run(function() {
        window.statusText.setText("상태: AI 자동 운전 중");
        window.statusText.setTextColor(colors.parseColor("#2ECC71"));
        window.btnToggle.setText("⏸ 정지");
        window.btnToggle.setBackgroundColor(colors.parseColor("#E74C3C"));
    });

    macroThread = threads.start(function() {
        toast("무한의 농장 스마트 AI 자동 플레이 시작!");
        while (isRunning) {
            // 1단계: 상자 깨기 & 작물 생성 (밭 전체 탭)
            breakBoxes();
            sleep(300);

            // 2단계: 번호 인식 및 동일 레벨 합성
            let merged = mergeSameNumbers();
            
            // 번호 인식이 없더라도 밭 사이를 가볍게 스와이프 보조
            if (!merged || merged === 0) {
                let rx1 = random(farmArea.left, farmArea.left + 200);
                let ry1 = random(farmArea.top, farmArea.bottom);
                swipe(rx1, ry1, rx1 + 150, ry1, 200);
            }
            sleep(400);
        }
    });
}

// 번호 인식 합성 전용 루프

    macroThread = threads.start(function() {
        toast("작물 번호 인식 자동 합성 모드 시작");

    });
}

function stopLoop() {
    isRunning = false;
    if (macroThread && macroThread.isAlive()) {
        macroThread.interrupt();
    }
    ui.run(function() {
        window.statusText.setText("상태: 대기 중");
        window.statusText.setTextColor(colors.parseColor("#E74C3C"));
        window.btnToggle.setText("▶ AI 시작");
        window.btnToggle.setBackgroundColor(colors.parseColor("#2ECC71"));
    });
}
